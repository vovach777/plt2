cmake_minimum_required(VERSION 3.14)
project(ctx4d CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# CTX4D_ARCH: native (default), avx2, ivybridge, ... ; on targets without AVX2 the code goes through SIMDe
set(CTX4D_ARCH "native" CACHE STRING "-march value (GCC/Clang)")
option(CTX4D_FORCE_SIMDE "use SIMDe even when the target has AVX2" OFF)
set(SIMDE_DIR "" CACHE PATH "folder that contains simde/x86/avx2.h (empty: download simde v0.8.2)")

if(SIMDE_DIR)
  set(SIMDE_INC ${SIMDE_DIR})
else()
  include(FetchContent)
  FetchContent_Declare(simde GIT_REPOSITORY https://github.com/simd-everywhere/simde.git GIT_TAG v0.8.2 GIT_SHALLOW TRUE)
  FetchContent_GetProperties(simde)
  if(NOT simde_POPULATED)
    FetchContent_Populate(simde)
  endif()
  set(SIMDE_INC ${simde_SOURCE_DIR})
endif()

foreach(t ctx4d_enc ctx4d_dec)
  add_executable(${t} ${t}.cpp)
  target_include_directories(${t} PRIVATE ${SIMDE_INC})
  if(MSVC)
    target_compile_options(${t} PRIVATE /O2 /arch:AVX2)
  else()
    target_compile_options(${t} PRIVATE -O3 -march=${CTX4D_ARCH})
  endif()
  if(CTX4D_FORCE_SIMDE)
    target_compile_definitions(${t} PRIVATE FORCE_SIMDE)
  endif()
endforeach()
