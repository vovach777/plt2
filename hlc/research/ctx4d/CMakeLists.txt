# ctx4d research kernels (4-D context encoder kernel + wavefront decoder); built from the plt2 root (PLT_BUILD_CTX4D)
# with the root PLT_ARCH. Without AVX2 in the target the bundled SIMDe (../../../avx2.h) is used.
option(CTX4D_FORCE_SIMDE "use SIMDe even when the target has AVX2" OFF)
# SIMDe on an AVX1-only target (Sandy/Ivy Bridge) is far slower than on plain SSE4 (256-bit integer ops emulated
# through float AVX); when the target has no AVX2, -mno-avx keeps SIMDe on its SSE path (gcc/clang only).
option(CTX4D_NO_AVX1 "add -mno-avx when the target lacks AVX2" ON)
set(CTX4D_EXTRA "")
if(CTX4D_NO_AVX1 AND (NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC" OR CMAKE_CXX_COMPILER_ID STREQUAL "Clang"))   # gcc, clang, clang-cl
  include(CheckCXXSourceCompiles)
  if(PLT_ARCH)
    set(CMAKE_REQUIRED_FLAGS "-march=${PLT_ARCH}")
  else()
    set(CMAKE_REQUIRED_FLAGS "-march=native")
  endif()
  check_cxx_source_compiles("#ifndef __AVX2__\n#error no avx2\n#endif\nint main(){return 0;}" CTX4D_TARGET_AVX2)
  unset(CMAKE_REQUIRED_FLAGS)
  if(NOT CTX4D_TARGET_AVX2)
    set(CTX4D_EXTRA -mno-avx -Wno-psabi)
    message(STATUS "ctx4d: target without AVX2 -> SIMDe on SSE (-mno-avx)")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      message(WARNING "ctx4d: gcc turns the SIMDe 256-bit integer emulation into element loops (~8x slower); use clang++ for this target")
    endif()
  endif()
endif()
if(NOT CTX4D_EXTRA)
  message(STATUS "ctx4d: ${CMAKE_CXX_COMPILER_ID} (${CMAKE_CXX_COMPILER_FRONTEND_VARIANT}), target AVX2: ${CTX4D_TARGET_AVX2}")
endif()
foreach(t ctx4d_enc ctx4d_dec ctx4d_vec)
  add_executable(${t} ${t}.cpp simd.h)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      target_compile_options(${t} PRIVATE /O2 -march=$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},native> ${CTX4D_EXTRA})
    else()
      target_compile_options(${t} PRIVATE /O2 /arch:$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},AVX2>)
    endif()
  else()
    target_compile_options(${t} PRIVATE -O3 -march=$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},native> ${CTX4D_EXTRA})
  endif()
  if(CTX4D_FORCE_SIMDE)
    target_compile_definitions(${t} PRIVATE FORCE_SIMDE)
  endif()
  set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
                        RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR} RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR})
endforeach()

# prototype codec (ctx4d kernels + hlc's FSE4); needs fse_huf from palic-ctx (plt2 root, PLT_BUILD_PCTX)
if(TARGET fse_huf)
  add_executable(ctx4d_codec ctx4d_codec.cpp ctx4d_vec.cpp)
  set_source_files_properties(ctx4d_vec.cpp PROPERTIES HEADER_FILE_ONLY ON)
  target_link_libraries(ctx4d_codec PRIVATE fse_huf)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(ctx4d_codec PRIVATE /O2 -march=$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},native> ${CTX4D_EXTRA})
  else()
    target_compile_options(ctx4d_codec PRIVATE -O3 -march=$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},native> ${CTX4D_EXTRA})
  endif()
  set_target_properties(ctx4d_codec PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
                        RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR} RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR})
  # experimental codec C4B1: independent compute blocks (default 128 x 16), fused in L1 (includes ctx4d_codec.cpp)
  add_executable(ctx4d_blk ctx4d_blk.cpp)
  target_link_libraries(ctx4d_blk PRIVATE fse_huf)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(ctx4d_blk PRIVATE /O2 -march=$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},native> ${CTX4D_EXTRA})
  else()
    target_compile_options(ctx4d_blk PRIVATE -O3 -march=$<IF:$<BOOL:${PLT_ARCH}>,${PLT_ARCH},native> ${CTX4D_EXTRA})
  endif()
  set_target_properties(ctx4d_blk PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
                        RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR} RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR})
endif()
