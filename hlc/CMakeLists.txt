cmake_minimum_required(VERSION 3.16)
project(hlc CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_executable(hlc hlc.cpp ../palic-1/rgb.hpp ../avx2.h)
if(NOT TARGET fse_huf)
  message(FATAL_ERROR "hlc is built from the plt2 root (needs fse_huf from palic-ctx)")
endif()
target_link_libraries(hlc PRIVATE fse_huf Threads::Threads)
set(HLC_ARCH "" CACHE STRING "target ISA (from the plt2 root PLT_ARCH is used)")
if(NOT HLC_ARCH AND PLT_ARCH)
  set(HLC_ARCH ${PLT_ARCH})
endif()
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
    target_compile_options(hlc PRIVATE /O3 $<IF:$<BOOL:${HLC_ARCH}>,/arch:${HLC_ARCH},/QxHost>)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    target_compile_options(hlc PRIVATE /O2 -march=$<IF:$<BOOL:${HLC_ARCH}>,${HLC_ARCH},native> -Wno-psabi)
  else()
    target_compile_options(hlc PRIVATE /O2 /arch:$<IF:$<BOOL:${HLC_ARCH}>,${HLC_ARCH},AVX>)
  endif()
else()
  target_compile_options(hlc PRIVATE -O3 -march=$<IF:$<BOOL:${HLC_ARCH}>,${HLC_ARCH},native> -Wno-psabi)
endif()
