cmake_minimum_required(VERSION 3.21)
project(PALIC CXX)
add_compile_definitions(_USE_MATH_DEFINES NOMINMAX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
# LLR_ARCH: gcc/clang/clang-cl -march value (default native); cl.exe: /arch value (default AVX). From the plt2 root PLT_ARCH is used.
set(LLR_ARCH "" CACHE STRING "target ISA")
if(NOT LLR_ARCH AND PLT_ARCH)          # built from the plt2 root: follow PLT_ARCH
  set(LLR_ARCH ${PLT_ARCH})
endif()

add_executable(p-conv main.cpp netpbm.hpp profiling.hpp rgb.hpp ../avx2.h)
add_executable(llrans llrans.cpp rgb.hpp ../avx2.h)
target_link_libraries(llrans PRIVATE Threads::Threads)
foreach(t p-conv llrans)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
      target_compile_options(${t} PRIVATE /O3 $<IF:$<BOOL:${LLR_ARCH}>,/arch:${LLR_ARCH},/QxHost>)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      target_compile_options(${t} PRIVATE /O2 -march=$<IF:$<BOOL:${LLR_ARCH}>,${LLR_ARCH},native> -Wno-psabi)
    else()
      target_compile_options(${t} PRIVATE /O2 /arch:$<IF:$<BOOL:${LLR_ARCH}>,${LLR_ARCH},AVX>)
    endif()
  else()
    target_compile_options(${t} PRIVATE -O3 -march=$<IF:$<BOOL:${LLR_ARCH}>,${LLR_ARCH},native> -Wno-psabi)
  endif()
endforeach()
