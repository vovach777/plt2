cmake_minimum_required(VERSION 3.16)
project(pctx C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(FSE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fse)
add_library(fse_huf STATIC ${FSE_DIR}/entropy_common.c ${FSE_DIR}/fse_compress.c ${FSE_DIR}/fse_decompress.c
                           ${FSE_DIR}/hist.c ${FSE_DIR}/huf_compress.c ${FSE_DIR}/huf_decompress.c ${FSE_DIR}/debug.c
                           ${FSE_DIR}/fseU16.c)
target_compile_definitions(fse_huf PUBLIC FSEU16_MAX_SYMBOL_VALUE=1023)
target_include_directories(fse_huf PUBLIC ${FSE_DIR})
add_executable(pctx pctx.cpp)
target_link_libraries(pctx PRIVATE fse_huf)
set(PCTX_ARCH "" CACHE STRING "target ISA (from the plt2 root PLT_ARCH is used)")
if(NOT PCTX_ARCH AND PLT_ARCH)
  set(PCTX_ARCH ${PLT_ARCH})
endif()
foreach(t pctx fse_huf)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
      target_compile_options(${t} PRIVATE /O3 $<IF:$<BOOL:${PCTX_ARCH}>,/arch:${PCTX_ARCH},/QxHost>)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      target_compile_options(${t} PRIVATE /O2 -march=$<IF:$<BOOL:${PCTX_ARCH}>,${PCTX_ARCH},native>)
    else()
      target_compile_options(${t} PRIVATE /O2 /arch:$<IF:$<BOOL:${PCTX_ARCH}>,${PCTX_ARCH},AVX>)
    endif()
  else()
    target_compile_options(${t} PRIVATE -O3 -march=$<IF:$<BOOL:${PCTX_ARCH}>,${PCTX_ARCH},native>)
  endif()
endforeach()
